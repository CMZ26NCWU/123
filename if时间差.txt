#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
int main() {
	int hour1, minute1;
	int hour2, minute2;
	printf("请输入开始时间:\n");
	scanf("%d %d",&hour1,&minute1);
	printf("请输入结束时间:\n");
	scanf("%d %d", &hour2, &minute2);
	int ih = hour2 - hour1;
	int im = minute2 - minute1;
	if (im < 0) {
		im = im + 60;
		ih--;
	}
	printf("过去%d小时，%d分钟",ih, im);

}
